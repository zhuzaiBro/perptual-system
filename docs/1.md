## 1. 多头和空头

**多头（Long）**：预期价格上涨，买入合约
**空头（Short）**：预期价格下跌，卖出合约

```solidity
// 仓位方向定义
enum Direction {
    LONG,   // 多头：价格涨你赚钱
    SHORT   // 空头：价格跌你赚钱
}

// 盈亏计算
function calcPnL(
    Direction direction,
    uint256 entryPrice,
    uint256 currentPrice,
    uint256 size          // 仓位大小（BTC数量）
) public pure returns (int256) {
    if (direction == Direction.LONG) {
        // 多头：现价 - 开仓价，涨了就是正数
        return int256(currentPrice - entryPrice) * int256(size);
    } else {
        // 空头：开仓价 - 现价，跌了就是正数
        return int256(entryPrice - currentPrice) * int256(size);
    }
}
```

```go
type Direction int

const (
    Long  Direction = 0
    Short Direction = 1
)

func CalcPnL(dir Direction, entryPrice, currentPrice, size float64) float64 {
    if dir == Long {
        return (currentPrice - entryPrice) * size
    }
    return (entryPrice - currentPrice) * size
}

// 例子：
// BTC 开仓价 $60000，现价 $65000，买了 1 BTC
// 多头盈亏 = (65000 - 60000) * 1 = +$5000 ✅

// BTC 开仓价 $60000，现价 $65000，卖出 1 BTC
// 空头盈亏 = (60000 - 65000) * 1 = -$5000 ❌
```

## 2. 合约价格 vs 市场价格 vs 资金费率

### 两个价格的区别

```
现货价格（Index Price）
  = 多个CEX现货市场的加权平均
  = BTC"真实"价值
  = Binance现货 * 30% + OKX现货 * 25% + ...

合约价格（Mark Price）  
  = 现货指数价格 + 基差EMA
  = 用来计算盈亏和清算的价格
  = 防止被单一交易所价格操纵
```

```go
// 现货指数价格
type PriceSource struct {
    Exchange string
    Price    float64
    Weight   float64
}

func CalcIndexPrice(sources []PriceSource) float64 {
    var totalWeight, weightedSum float64
    for _, s := range sources {
        weightedSum += s.Price * s.Weight
        totalWeight += s.Weight
    }
    return weightedSum / totalWeight
}

// 标记价格 = 现货指数 + 基差EMA
// 基差 = 合约市场价 - 现货指数价
// EMA平滑处理防止瞬间波动
func CalcMarkPrice(indexPrice, basisEMA float64) float64 {
    return indexPrice + basisEMA
}
```

### 资金费率

**核心逻辑：让合约价格锚定现货价格**

```
多头多 → 合约价 > 现货价 → 多头付钱给空头（多头被惩罚）
空头多 → 合约价 < 现货价 → 空头付钱给多头（空头被惩罚）
```

```go
// 资金费率计算（每8小时结算一次）
type FundingRate struct {
    InterestRate float64 // 固定利率，通常 0.01%
    PremiumIndex float64 // 溢价指数
}

func CalcFundingRate(markPrice, indexPrice float64) float64 {
    // 溢价指数 = (合约价 - 现货价) / 现货价
    premiumIndex := (markPrice - indexPrice) / indexPrice

    // 固定利率（BTC通常 0.01%/8h）
    interestRate := 0.0001

    // 资金费率 = 溢价指数 + clamp(利率 - 溢价指数, -0.05%, 0.05%)
    diff := interestRate - premiumIndex
    diff = math.Max(-0.0005, math.Min(0.0005, diff)) // clamp
    return premiumIndex + diff
}

func CalcFundingPayment(
    fundingRate float64,
    positionSize float64, // BTC数量
    markPrice float64,
    isLong bool,
) float64 {
    // 资金费用 = 费率 * 仓位名义价值
    payment := fundingRate * positionSize * markPrice

    // 费率为正：多头付给空头
    // 费率为负：空头付给多头
    if isLong {
        return -payment // 多头付出（负数）
    }
    return payment // 空头收入（正数）
}
```

```
具体例子：
  BTC 合约价 $61000，现货价 $60000
  溢价 = (61000-60000)/60000 = 1.67%  → 多头偏多

  资金费率 ≈ +0.01%（正值，多头付）

  你持有 1 BTC 多头，名义价值 $61000
  每8小时付出 = 0.0001 * 1 * 61000 = $6.1
```

## 3. 保证金和仓位

### 基本概念

```
名义价值（Notional）= 你实际控制的仓位大小
保证金（Margin）   = 你实际投入的钱
杠杆（Leverage）   = 名义价值 / 保证金
```

```go
type Position struct {
    Trader     string
    Direction  Direction
    Size       float64 // BTC数量（名义仓位）
    EntryPrice float64 // 开仓价
    Margin     float64 // 已投入保证金（USDT）
    Leverage   float64 // 杠杆倍数
}

func NewPosition(margin, leverage, entryPrice float64, dir Direction) Position {
    // 名义价值 = 保证金 * 杠杆
    notional := margin * leverage
    // BTC数量 = 名义价值 / 开仓价
    size := notional / entryPrice

    return Position{
        Size:       size,
        EntryPrice: entryPrice,
        Margin:     margin,
        Leverage:   leverage,
        Direction:  dir,
    }
}

// 例子：
// 投入 $1000 保证金，10x杠杆，BTC价格 $60000
// 名义价值 = $1000 * 10 = $10000
// 控制 BTC = $10000 / $60000 = 0.1667 BTC
```

### 合约层：保证金和清算逻辑

```solidity
struct Position {
    address trader;
    bool    isLong;
    uint256 size;        // 仓位大小（以USD计，1e18精度）
    uint256 entryPrice;  // 开仓价格
    uint256 margin;      // 保证金（USDT）
    uint256 lastFundingIndex; // 上次结算资金费率时的index
}

mapping(address => Position) public positions;

// 开仓
function openPosition(
    bool isLong,
    uint256 margin,      // 投入保证金
    uint256 leverage,    // 杠杆倍数
    uint256 markPrice    // 当前标记价格
) external {
    uint256 notional = margin * leverage;       // 名义价值
    uint256 size = notional * 1e18 / markPrice; // BTC数量

    positions[msg.sender] = Position({
        trader:           msg.sender,
        isLong:           isLong,
        size:             size,
        entryPrice:       markPrice,
        margin:           margin,
        lastFundingIndex: currentFundingIndex
    });

    // 收取保证金
    USDT.transferFrom(msg.sender, address(this), margin);
}

// 保证金率 = (保证金 + 未实现盈亏) / 名义价值
function getMarginRatio(
    address trader,
    uint256 currentPrice
) public view returns (uint256) {
    Position memory pos = positions[trader];
    
    // 未实现盈亏
    int256 unrealizedPnL;
    if (pos.isLong) {
        unrealizedPnL = int256(currentPrice - pos.entryPrice) 
                      * int256(pos.size) / 1e18;
    } else {
        unrealizedPnL = int256(pos.entryPrice - currentPrice) 
                      * int256(pos.size) / 1e18;
    }

    // 当前净保证金
    int256 netMargin = int256(pos.margin) + unrealizedPnL;
    
    // 名义价值
    uint256 notional = pos.size * currentPrice / 1e18;

    return uint256(netMargin) * 1e18 / notional;
}

// 清算判断：保证金率 < 维持保证金率（比如5%）
function isLiquidatable(
    address trader,
    uint256 currentPrice
) public view returns (bool) {
    uint256 marginRatio = getMarginRatio(trader, currentPrice);
    return marginRatio < MAINTENANCE_MARGIN_RATIO; // 5% = 5e16
}
```

## 整体数据流串联

```
用户投入 $1000 保证金，10x杠杆，做多BTC @ $60000
  ↓
size = $10000 / $60000 = 0.1667 BTC
  ↓
BTC涨到 $65000
  ↓
未实现盈亏 = (65000 - 60000) * 0.1667 = +$833

保证金率  = $1833 / ($65000 * 0.1667) = 16.9%  → 安全

BTC跌到 $54500
  ↓
未实现盈亏 = (54500 - 60000) * 0.1667 = -$917
净保证金  = $1000 - $917 = $83
保证金率  = $83 / ($54500 * 0.1667) = 0.91%  → 触发清算 ❌
```

### 资金费率对保证金的影响

```
净保证金 = 初始保证金 + 未实现盈亏 - 累计资金费用
```

#### Go：资金费率累计计算

```go
type Position struct {
    Trader     string
    Direction  Direction
    Size       float64   // BTC数量
    EntryPrice float64   // 开仓价
    Margin     float64   // 初始保证金
    Leverage   float64

    // 资金费率相关
    EntryFundingIndex float64 // 开仓时的累计资金费率index
}

type FundingState struct {
    CumulativeIndex float64 // 全局累计资金费率index（每8h累加）
    LastUpdateTime  int64
}

// 每8小时更新一次全局index
func (fs *FundingState) Update(fundingRate float64) {
    // index 累加，类似利息复利
    fs.CumulativeIndex += fundingRate
    fs.LastUpdateTime = time.Now().Unix()
}

// 计算该仓位累计欠的资金费用
func CalcAccumulatedFunding(
    pos Position,
    currentFundingIndex float64,
    currentPrice float64,
) float64 {
    // 费率差 = 从开仓到现在，累计了多少费率
    fundingDelta := currentFundingIndex - pos.EntryFundingIndex

    // 名义价值
    notional := pos.Size * currentPrice

    // 资金费用 = 费率差 * 名义价值
    fundingCost := fundingDelta * notional

    // 多头：费率为正时付出（负收益）
    // 空头：费率为正时收入（正收益）
    if pos.Direction == Long {
        return -fundingCost
    }
    return fundingCost
}

// 计算净保证金（考虑盈亏 + 资金费率）
func CalcNetMargin(
    pos Position,
    currentPrice float64,
    currentFundingIndex float64,
) float64 {
    // 未实现盈亏
    pnl := CalcPnL(pos.Direction, pos.EntryPrice, currentPrice, pos.Size)

    // 累计资金费用
    funding := CalcAccumulatedFunding(pos, currentFundingIndex, currentPrice)

    return pos.Margin + pnl + funding
}

// 保证金率
func CalcMarginRatio(
    pos Position,
    currentPrice float64,
    currentFundingIndex float64,
) float64 {
    netMargin := CalcNetMargin(pos, currentPrice, currentFundingIndex)
    notional := pos.Size * currentPrice
    return netMargin / notional
}
```

#### Solidity：链上资金费率结算

```solidity
contract PerpEngine {

    // 全局累计资金费率 index（每8h更新）
    // 正值：多头付给空头
    // 负值：空头付给多头
    int256 public globalFundingIndex;
    uint256 public lastFundingTime;
    uint256 constant FUNDING_INTERVAL = 8 hours;

    struct Position {
        address trader;
        bool    isLong;
        uint256 size;               // BTC数量 1e18精度
        uint256 entryPrice;
        uint256 margin;             // 初始保证金 USDT
        int256  entryFundingIndex;  // 开仓时记录的全局index
    }

    mapping(address => Position) public positions;

    // ── 每8小时调用一次，更新全局资金费率 index ──
    function updateFunding(int256 fundingRate) external {
        require(block.timestamp >= lastFundingTime + FUNDING_INTERVAL);
        globalFundingIndex += fundingRate;
        lastFundingTime = block.timestamp;
    }

    // ── 开仓时记录当前index ──
    function openPosition(
        bool isLong,
        uint256 margin,
        uint256 leverage,
        uint256 markPrice
    ) external {
        uint256 notional = margin * leverage;
        uint256 size = notional * 1e18 / markPrice;

        positions[msg.sender] = Position({
            trader:            msg.sender,
            isLong:            isLong,
            size:              size,
            entryPrice:        markPrice,
            margin:            margin,
            entryFundingIndex: globalFundingIndex  // 记录开仓时的index
        });

        USDT.transferFrom(msg.sender, address(this), margin);
    }

    // ── 计算累计资金费用 ──
    // 返回值：正数=收入，负数=付出
    function calcFundingPayment(
        address trader,
        uint256 currentPrice
    ) public view returns (int256) {
        Position memory pos = positions[trader];

        // 从开仓到现在累计的费率差
        int256 fundingDelta = globalFundingIndex - pos.entryFundingIndex;

        // 名义价值
        uint256 notional = pos.size * currentPrice / 1e18;

        // 资金费用
        int256 fundingCost = fundingDelta * int256(notional) / 1e18;

        // 多头付出，空头收入
        if (pos.isLong) {
            return -fundingCost;
        }
        return fundingCost;
    }

    // ── 净保证金 = 初始保证金 + 盈亏 + 资金费收支 ──
    function getNetMargin(
        address trader,
        uint256 currentPrice
    ) public view returns (int256) {
        Position memory pos = positions[trader];

        // 未实现盈亏
        int256 pnl;
        if (pos.isLong) {
            pnl = int256(currentPrice - pos.entryPrice)
                * int256(pos.size) / 1e18;
        } else {
            pnl = int256(pos.entryPrice - currentPrice)
                * int256(pos.size) / 1e18;
        }

        // 资金费收支
        int256 funding = calcFundingPayment(trader, currentPrice);

        return int256(pos.margin) + pnl + funding;
    }

    // ── 保证金率 ──
    function getMarginRatio(
        address trader,
        uint256 currentPrice
    ) public view returns (int256) {
        int256 netMargin = getNetMargin(trader, currentPrice);
        uint256 notional = positions[trader].size * currentPrice / 1e18;
        return netMargin * 1e18 / int256(notional);
    }

    // ── 清算判断 ──
    function isLiquidatable(
        address trader,
        uint256 currentPrice
    ) public view returns (bool) {
        int256 marginRatio = getMarginRatio(trader, currentPrice);
        // 保证金率 < 5% 或净保证金为负
        return marginRatio < 5e16 || getNetMargin(trader, currentPrice) < 0;
    }
}
```

### 完整数字例子

```
开仓：$1000 保证金，10x杠杆，做多BTC @ $60000
  size = 0.1667 BTC
  entryFundingIndex = 0

经过3个周期（24小时），每期资金费率 +0.01%
  globalFundingIndex = 0.0003

BTC现价 $58000（跌了）

── 未实现盈亏 ──
pnl = (58000 - 60000) * 0.1667 = -$333.4

── 累计资金费用（多头付出）──
fundingDelta  = 0.0003 - 0 = 0.0003
notional      = 0.1667 * 58000 = $9668.6
fundingCost   = 0.0003 * 9668.6 = $2.9（付出）

── 净保证金 ──
netMargin = $1000 - $333.4 - $2.9 = $663.7

── 保证金率 ──
marginRatio = $663.7 / $9668.6 = 6.87%  → 暂时安全

── 如果BTC继续跌到 $54000 ──
pnl         = (54000 - 60000) * 0.1667 = -$1000.2
fundingCost ≈ $2.7（名义价值变小了）
netMargin   = $1000 - $1000.2 - $2.7 = -$2.9  → 净保证金为负 → 强制清算
```

资金费率单次影响小，但持仓时间长、费率持续为正的情况下会**持续侵蚀保证金**，极端行情叠加高资金费率会加速清算。

## 平仓时的钱从哪来、到哪去

```
平仓结算 = 初始保证金 + 未实现盈亏 - 累计资金费用 - 手续费

每一项都可能是正负
```

#### 场景设定
```
开仓：$1000 保证金，10x杠杆，做多BTC @ $60000
```

#### 情况A：赚钱了，BTC涨到 $65000 平仓

### Solidity：平仓结算逻辑

```solidity
function closePosition(uint256 markPrice) external {
    Position memory pos = positions[msg.sender];
    require(pos.size > 0, "no position");

    // 1. 未实现盈亏
    int256 pnl;
    if (pos.isLong) {
        pnl = int256(markPrice - pos.entryPrice)
            * int256(pos.size) / 1e18;
    } else {
        pnl = int256(pos.entryPrice - markPrice)
            * int256(pos.size) / 1e18;
    }

    // 2. 累计资金费用
    int256 funding = calcFundingPayment(msg.sender, markPrice);

    // 3. 平仓手续费（名义价值 * 费率）
    uint256 notional = pos.size * markPrice / 1e18;
    uint256 closeFee = notional * TAKER_FEE_RATE / 1e18; // 0.05%

    // 4. 最终结算金额
    int256 settlement = int256(pos.margin) + pnl + funding - int256(closeFee);

    // 5. 清除仓位
    delete positions[msg.sender];

    // 6. 打款
    if (settlement > 0) {
        // 有钱拿
        USDT.transfer(msg.sender, uint256(settlement));
        // 手续费进协议金库
        USDT.transfer(feeVault, closeFee);
    } else {
        // 亏完了（理论上主动平仓不会出现，被清算才会）
        // 保证金全部归协议处理
    }

    emit PositionClosed(msg.sender, pnl, funding, closeFee, settlement);
}
```

### Go：后端计算平仓预览

```go
type CloseResult struct {
    PnL        float64 // 盈亏
    Funding    float64 // 资金费收支（负=付出）
    CloseFee   float64 // 手续费（付出）
    Settlement float64 // 最终到账
}

func CalcCloseResult(
    pos        Position,
    closePrice float64,
    fundingIdx float64,
    feeRate    float64, // 0.0005
) CloseResult {
    // 盈亏
    pnl := CalcPnL(pos.Direction, pos.EntryPrice, closePrice, pos.Size)

    // 资金费
    funding := CalcAccumulatedFunding(pos, fundingIdx, closePrice)

    // 手续费
    notional := pos.Size * closePrice
    fee := notional * feeRate

    // 结算
    settlement := pos.Margin + pnl + funding - fee

    return CloseResult{
        PnL:        pnl,
        Funding:    funding,
        CloseFee:   -fee,
        Settlement: settlement,
    }
}
```

### 汇总：平仓能赚/亏哪些钱

```
┌─────────────────┬──────────────────────────────┬──────────┐
│ 项目            │ 说明                          │ 方向     │
├─────────────────┼──────────────────────────────┼──────────┤
│ 初始保证金      │ 你投入的本金                  │ +（返还）│
│ 未实现盈亏      │ 价差 * size                   │ +/-      │
│ 累计资金费用    │ 持仓期间每8h结算              │ +/-      │
│ 平仓手续费      │ 名义价值 * Taker费率          │ -（付出）│
└─────────────────┴──────────────────────────────┴──────────┘

主动平仓 vs 被清算的区别：
  主动平仓：你决定时机，settlement 通常还有剩余
  被清算：  系统强制，还要额外扣清算罚金（比如1%名义价值）
            剩余才进保险基金，你拿不到
```
